def DES_encrypt(plaintext, key):

    """
    Encrypts the given plaintext using the DES algorithm.

    Parameters:
    plaintext (str): The plaintext to be encrypted.
    key (list): The key to be used for subkey generation for DES.

    Returns:
    str: The ciphertext generated from the encryption process.
    """

    # Generate subkeys from the input key
    subkeys = generate_subkeys(key)

    # Apply IP
    permuted_data = IP(plaintext)   # initial permutation
    left, right = [0] * (NUM_ROUNDS+1), [0] * (NUM_ROUNDS+1)     
    left[0], right[0] = permuted_data[:32], permuted_data[32:]
    
    # 16 rounds
    for round in range(1, NUM_ROUNDS+1):
        temp = XOR( F(right[round - 1], subkeys[round - 1]), left[round - 1])
        left[round], right[round] = right[round - 1], temp
    
    # Combine and apply FP
    combined_data = right[NUM_ROUNDS] + left[NUM_ROUNDS]     # NUM_ROUNDS = 16
    ciphertext = FP(combined_data)  # final permutation to obtain the output ciphertext

    return ciphertext
