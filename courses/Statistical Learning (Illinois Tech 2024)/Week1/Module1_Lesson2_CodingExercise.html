# Step 2: Define five functions
def f1(x): return 2*x + 6
def f2(x): return 4*x + 4
def f3(x): return 3*x + 5
def f4(x): return x + 8
def f5(x): return 5*x + 3

functions = [f1, f2, f3, f4, f5]

We then compute the mean squared error. Notice that we compute the squared errors: the predictions minus the actua values squared. It becomes a mean when we divide by the number of datapoints (len(Y)).

In [10]:
# Step 3: Compute MSE for each function
def compute_mse(f, X, Y):
    predictions = f(X)
    return np.mean((predictions - Y)**2) / len(Y)

mse_list = []
for func in functions:
    mse = compute_mse(func, X, Y)
    mse_list.append(mse)

We now plot the lines. The colors are ordered according to the order of the functions, so f1 corresponds to red, f2 corresponds to blue and so on. When we plot the lines, we are also plotting the mean squared error as a label from the previous step. This will display in the legend.

# Step 4: Plot the functions, dataset, and MSEs
plt.figure(figsize=(10,6))
plt.scatter(X, Y, color='black', label='Data points')

colors = ['red', 'blue', 'green', 'purple', 'orange']

for idx, (func, color) in enumerate(zip(functions, colors)):
    plt.plot(X, func(X), color=color, label=f'f{idx+1}, MSE: {mse_list[idx]:.2f}')
    
plt.legend()
plt.title("Functions and their Mean Squared Error with the Dataset")
plt.xlabel("X (Independent Variable)")
plt.ylabel("Y (Dependent Variable)")
plt.grid(True)
plt.show()
